import csv
import io


def run(arguments):
    csv_text = arguments.get("csv", "")
    column = arguments.get("column")
    if not isinstance(csv_text, str):
        csv_text = "" if csv_text is None else str(csv_text)
    f = io.StringIO(csv_text)
    reader = csv.DictReader(f)
    if reader.fieldnames is None:
        return {"count": 0, "sum": 0.0, "mean": None, "min": None, "max": None}
    if column not in reader.fieldnames:
        raise ValueError("column not found in CSV header")
    count = 0
    total = 0.0
    minimum = None
    maximum = None
    for row in reader:
        raw = row.get(column)
        if raw is None:
            continue
        if isinstance(raw, str):
            raw = raw.strip()
            if raw == "":
                continue
        value = float(raw)
        count += 1
        total += value
        if minimum is None or value < minimum:
            minimum = value
        if maximum is None or value > maximum:
            maximum = value
    if count == 0:
        return {"count": 0, "sum": 0.0, "mean": None, "min": None, "max": None}
    mean = total / count
    return {"count": count, "sum": float(total), "mean": float(mean), "min": float(minimum), "max": float(maximum)}
