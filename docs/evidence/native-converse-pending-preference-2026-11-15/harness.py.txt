class Wire:
 def __init__(self,connection,file): self.connection,self.file=connection,file
 def record(self,direction,raw):
  assert type(raw) is str and len(raw.encode())<=256*1024
  self.file.write(json.dumps({'direction':direction,'event':json.loads(raw)},ensure_ascii=False)+'\n');self.file.flush()
 async def send(self,raw):
  self.record('send',raw);return await self.connection.send(raw)
 async def recv(self):
  raw=await self.connection.recv();self.record('recv',raw);return raw
 async def close(self):
  try: return await self.connection.close()
  finally: self.file.close()
