"""#623: actual fresh native HTTP/IPC and Dart decoder rollover ownership."""

import asyncio
import json
import os
import shutil
import socket
from pathlib import Path
from urllib.parse import urljoin

import pytest
import uvicorn
from flutter_project_helpers import isolated_flutter_project

from tests.integration.test_incremental_whistle_converse import (
    converse_incremental_env as converse_incremental_env,
)
from tests.integration.test_live_stream_api import live_client, scope_for


@pytest.mark.parametrize(
    "entrypoint", ["live_stream_native_live_test.dart", "adaptive_live_call_native_test.dart"]
)
async def test_actual_dart_native_live_stream_rollover(
    converse_incremental_env, tmp_path, entrypoint
):
    flutter = shutil.which("flutter")
    root = Path(__file__).resolve().parents[2]
    cached = root / "apps/kinaiya_flutter/.dart_tool"
    if flutter is None or not all(
        (cached / name).is_file() for name in ("package_config.json", "package_graph.json")
    ):
        pytest.skip("Pinned Flutter SDK and existing dependency cache required; no installation")
    project = isolated_flutter_project(root, tmp_path / "native-stream-flutter")
    prepared = project / ".dart_tool"
    prepared.mkdir()
    package_config = json.loads((cached / "package_config.json").read_text())
    for package in package_config["packages"]:
        package["rootUri"] = (
            project.as_uri() + "/"
            if package["name"] == "kinaiya_flutter"
            else urljoin((cached / "package_config.json").as_uri(), package["rootUri"])
        )
    (prepared / "package_config.json").write_text(json.dumps(package_config))
    shutil.copyfile(cached / "package_graph.json", prepared / "package_graph.json")
    original_lock = (project / "pubspec.lock").read_bytes()
    requests = []
    env = converse_incremental_env
    async with live_client(env) as native:

        @native.app.middleware("http")
        async def observe(request, call_next):
            result = await call_next(request)
            requests.append((request.method, request.url.path, result.status_code))
            return result

        sock = socket.socket()
        sock.bind(("127.0.0.1", 0))
        sock.listen(16)
        port = sock.getsockname()[1]
        # live_client owns the actual app lifespan; the HTTP facade must not start it twice.
        server = uvicorn.Server(uvicorn.Config(native.app, lifespan="off", log_level="error"))
        actual = asyncio.create_task(server.serve(sockets=[sock]))
        process = None
        try:
            async with asyncio.timeout(3):
                while not server.started:
                    if actual.done():
                        await actual
                    await asyncio.sleep(0)
            process = await asyncio.create_subprocess_exec(
                flutter,
                "test",
                "--no-pub",
                "--reporter",
                "expanded",
                "test/" + entrypoint,
                cwd=project,
                env={
                    **os.environ,
                    "KINAIYA_LIVE_STREAM_URL": f"http://127.0.0.1:{port}",
                    "KINAIYA_LIVE_STREAM_TOKEN": "o" * 40,
                    "KINAIYA_LIVE_STREAM_SCOPE": json.dumps(scope_for(env)),
                },
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.STDOUT,
            )
            async with asyncio.timeout(90):
                output, _ = await process.communicate()
            (tmp_path / "native-live-stream-dart.log").write_bytes(output)
            assert process.returncode == 0, output.decode(errors="replace")
            assert b"All tests passed!" in output
            assert (project / "pubspec.lock").read_bytes() == original_lock
            if entrypoint == "adaptive_live_call_native_test.dart":
                live_requests = [
                    (method, path, status)
                    for method, path, status in requests
                    if path.startswith("/v2/converse/audio/live-stream/")
                ]
                assert live_requests and all(status == 200 for _, _, status in live_requests)
                for phase, count in (
                    ("begin", 1),
                    ("pcm", 2),
                    ("finish", 0),
                    ("cancel", 0),
                    ("cancel-original", 1),
                ):
                    assert (
                        sum(
                            method == "POST" and path.endswith("/" + phase)
                            for method, path, _ in live_requests
                        )
                        == count
                    )
                captures = list(native.app.state.live_streams.captures.values())
                assert len(captures) == 1
                capture = captures[0]
                assert capture.capture_fenced and capture.cleanup_complete()
                assert capture.native.lease.settled and not env.runtime.stt_model_settings.leases
                assert env.engine.begin_calls == 1 and env.engine.finish_calls == 0
                return
            assert all(status == 200 for _, _, status in requests)
            for phase, count in (
                ("begin", 2),
                ("next-utterance", 1),
                ("pcm", 2),
                ("finish", 2),
                ("cancel", 1),
                ("cancel-original", 1),
            ):
                assert (
                    sum(
                        method == "POST" and path.endswith("/" + phase)
                        for method, path, _ in requests
                    )
                    == count
                )
            assert (
                sum(
                    method == "GET" and path.endswith("/next-utterance")
                    for method, path, _ in requests
                )
                == 1
            )
            capture = native.app.state.live_streams.captures["d" * 32]
            assert (
                sum(
                    method == "GET" and path.endswith("/cancel-original")
                    for method, path, _ in requests
                )
                == 1
            )
            assert (
                sum(
                    method == "POST" and path.endswith("/cancel-original/lookup")
                    for method, path, _ in requests
                )
                == 1
            )
            assert capture.capture_fenced and capture.capture_task.done()
            assert capture.native.lease.settled
            assert not env.runtime.stt_model_settings.leases
            assert env.engine.begin_calls == 3 and env.engine.finish_calls == 2
        finally:
            if process is not None and process.returncode is None:
                process.kill()
                await process.wait()
            server.should_exit = True
            await asyncio.wait_for(asyncio.shield(actual), 5)
            sock.close()
