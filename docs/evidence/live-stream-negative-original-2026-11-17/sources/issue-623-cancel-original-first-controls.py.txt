"""#623: a negative original must fence even a start that has not arrived."""

import asyncio

import pytest

from packages.contracts.stt_settlement import digest
from services.converse.live_stream_api import BASE
from tests.integration.test_incremental_whistle_converse import AUTH
from tests.integration.test_incremental_whistle_converse import (
    converse_incremental_env as converse_incremental_env,
)
from tests.integration.test_live_stream_api import begin_body, live_client


def abort_body(initial, reason="cancelled"):
    return {
        "schema_version": "1.0",
        "protocol": "native_live_abort_v1",
        "initial": initial,
        "reason": reason,
    }


async def initial_body(env, live):
    owner = (await live.client.get(BASE + "/ownership")).json()["owner_instance"]
    return begin_body(env, owner)


def confirmed(proof, original):
    assert proof["original"] == original and proof["request_sha256"] == digest(original)
    assert proof["outcome"] == "confirmed" and proof["cleanup_confirmed"]
    assert not proof["capture"]["capture_open"] and proof["capture"]["fence"] is not None
    assert proof["capture"]["original"]["capture_id"] == original["initial"]["capture_id"]


async def test_negative_original_precedes_late_begin_without_registration_or_stt(
    converse_incremental_env,
):
    env = converse_incremental_env
    async with live_client(env) as live:
        initial = await initial_body(env, live)
        original = abort_body(initial)
        missing = await live.client.post(BASE + "/cancel-original/lookup", json=original)
        assert missing.status_code == 200 and missing.json()["outcome"] == "unknown"
        assert live.app.state.live_streams.captures == {}
        assert not env.runtime.stt_model_settings.leases and env.engine.begin_calls == 0
        # The existing identity-only cancellation contract is unchanged.
        legacy = {
            key: initial[key]
            for key in ("schema_version", "protocol", "owner_instance", "capture_id")
        }
        assert (
            await live.client.post(BASE + "/cancel", json={**legacy, "reason": "cancelled"})
        ).status_code == 409
        stopped = await live.client.post(BASE + "/cancel-original", json=original)
        assert stopped.status_code == 200
        confirmed(stopped.json(), original)
        assert stopped.json()["capture"]["receipt"] is None
        assert (await live.client.post(BASE + "/begin", json=initial)).status_code == 409
        confirmed(
            (await live.client.post(BASE + "/cancel-original/lookup", json=original)).json(),
            original,
        )
        assert not env.runtime.stt_model_settings.leases and env.engine.begin_calls == 0
        assert env.store.read(kinds=["conversation.window"]) == []


async def test_negative_original_holds_actual_native_fence_and_deduplicates_delivery(
    converse_incremental_env,
    monkeypatch,
):
    env = converse_incremental_env
    async with live_client(env) as live:
        original = abort_body(await initial_body(env, live))
        entered, release = asyncio.Event(), asyncio.Event()
        fence = env.journal.fence_live_capture
        calls = []

        async def held(request):
            calls.append(request)
            entered.set()
            await release.wait()
            return await fence(request)

        monkeypatch.setattr(env.journal, "fence_live_capture", held)
        operation = asyncio.create_task(live.client.post(BASE + "/cancel-original", json=original))
        try:
            await asyncio.wait_for(entered.wait(), 3)
            first = await asyncio.wait_for(asyncio.shield(operation), 3)
            assert first.status_code == 200
            assert (
                first.json()["outcome"] == "in_progress" and not first.json()["cleanup_confirmed"]
            )
            duplicate = await live.client.post(BASE + "/cancel-original", json=original)
            assert duplicate.status_code == 200 and not duplicate.json()["cleanup_confirmed"]
            assert len(calls) == 1
            assert (
                await live.client.post(BASE + "/begin", json=original["initial"])
            ).status_code == 409
            assert env.engine.begin_calls == 0 and not env.runtime.stt_model_settings.leases
        finally:
            release.set()
            await asyncio.wait_for(asyncio.shield(operation), 3)
        owner = live.app.state.live_streams.captures[original["initial"]["capture_id"]]
        await asyncio.wait_for(asyncio.shield(owner.mic.fence_task), 3)
        confirmed(
            (await live.client.post(BASE + "/cancel-original/lookup", json=original)).json(),
            original,
        )


async def test_negative_original_during_held_registration_preserves_no_stt_effect(
    converse_incremental_env,
    monkeypatch,
):
    env = converse_incremental_env
    async with live_client(env) as live:
        initial = await initial_body(env, live)
        entered, release = asyncio.Event(), asyncio.Event()
        register = env.journal.register_live_capture

        async def held(request):
            entered.set()
            await release.wait()
            return await register(request)

        monkeypatch.setattr(env.journal, "register_live_capture", held)
        operation = asyncio.create_task(live.client.post(BASE + "/begin", json=initial))
        try:
            await asyncio.wait_for(entered.wait(), 3)
            original = abort_body(initial, "disconnected")
            stopped = await live.client.post(BASE + "/cancel-original", json=original)
            assert stopped.status_code == 200 and not stopped.json()["cleanup_confirmed"]
            assert not stopped.json()["capture"]["capture_open"]
            assert stopped.json()["capture"]["fence"] is not None
            assert not env.runtime.stt_model_settings.leases and env.engine.begin_calls == 0
        finally:
            release.set()
            started = await asyncio.wait_for(asyncio.shield(operation), 3)
        assert started.status_code in (409, 503)
        owner = live.app.state.live_streams.captures[initial["capture_id"]]
        await asyncio.wait_for(asyncio.gather(*owner.tasks(), return_exceptions=True), 3)
        confirmed(
            (await live.client.post(BASE + "/cancel-original/lookup", json=original)).json(),
            original,
        )
        assert not env.runtime.stt_model_settings.leases and env.engine.begin_calls == 0


@pytest.mark.parametrize("role", [None, "agent", "converse", "decisions", "resources"])
@pytest.mark.parametrize("suffix", ["/cancel-original", "/cancel-original/lookup"])
async def test_negative_original_requires_owner_before_any_metadata_effect(
    converse_incremental_env,
    role,
    suffix,
):
    env = converse_incremental_env
    headers = {"Authorization": "" if role is None else "Bearer " + AUTH.tokens[role]}
    result = await env.client.post(BASE + suffix, headers=headers, json={})
    assert result.status_code == (401 if role is None else 403)
    assert env.engine.begin_calls == 0 and env.app.state.live_streams.captures == {}
